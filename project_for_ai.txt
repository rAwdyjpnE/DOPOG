# Как работать с репозиторием

## Как собрать проект
- Скачать godot 4.5.1
- Клонировать репозиторий
- Запустить godot и выбрать в нем DOPOG

## Общии правила
- Все что мы делаем находится в папке src
- В src/Doc находится документация проекта
- В src/Models находятся модели
- В src/Nodes находятся объекты godot
- В src/Scenes находятся сцены godot
- В src/Scripts находятся скрипты godot
- В src/Textures находятся текстуры

## Правила написания скриптов
- Используются двойные кавычки
- Константы пишутся капсом
- Названия функции, которые возвращают тип void, начинаются с "_" 

## Как писать документацию
- Документация пишется на русском языке
- Файл в документации называется также как и скрипт
- Сначала описывается общий смысл скрипта, потом переменных, потом функции
- При описании переменных указывается: const/var название переменной `класс` - общее описание смысла переменной
- При описании функции указывается: название(передаваемые аргументы) - общее описании смысла функции

## Коммиты в git
- Версия пишется такая же как и в последнем коммите, новая версия применяется только по соглашению со всей командой
- Описание коммита пишется полностью на английском языке и описывает основные изменения в коде
- При отправки изменений создайте свою ветку с названием вашего ника в тг. В конце недели, будем брать изменений в main ветку

# Button_stickers.gd

## Данный скрипт отвечает за смену стикера при нажатии кнопки

### Переменные

- var **link** `String` - хранит ссылку на текстуру для стикера
- var **main_script_link** `Node3D` - ссылка на основной скрипт `Input.gd`

### Функции

- **_ready() -> void** - задает кнопке функцию **on_button_pressed**
- **_on_button_pressed -> void** - при клике на кнопку меняет в скрипте `Input.gd` переменную `link_sticker`, чтобы поменять текстуру для стикера

# Input.gd

## обрабатывает ввод пользователя

### Переменные

- const **object** `Node3D` - ссылка основной объект на сцене
- var **camera** `Camera3D` - ссылка на основную камеру на сцене
- var **stickers** `Array` - массив, который хранит стикеры размещенные на сцене
- var **stickers_lenght** `int` - количество стикеров в массиве **stickers**
- var **link_sticker** `String` - хранит ссылку на путь текстуры стикера
- var **preview_sticker** `Decal` - хранит ссылку на декаль, который показывает как будет размещен стикер на объекте
- const **RIGHT** `str` - поворот вправо
- const **LEFT** `str` - поворот влево
- const **UP** `str` - поворот вверх
- const **DOWN** `str` - поворот вниз
- const **SPEED** `int` - скорость поворота объекта
- const **RAY_LENGTH** `int` - длина луча, который выпускается из камеры для проверки, возможно ли поставить стикер
- const **ROTATE_TIME** `float` - время, за которое объект поворачивается на 90 градусов
- var **is_rotating** `bool` - true если объект поворачивается, false в ином случае

### Функции

- **_input(event) -> void** - обрабатывает ввод пользователя
- **_rotate_object(axis, degrees) -> void** - поворачивает объект на указанное количество градусов вокруг заданной оси
- **_place_sticker() -> void** - размещает стикер на объекте
- **create_sticker() -> Decal** - создает стикер для дальнейшего размещения его на объекте
- **_delete_last_sticker() -> void** - удаляет последний стикер, который был на объекте
- **_safe_look_at(node: Node3D, pos: Vector3, normal: Vector3) -> void** - безопасный поворот стикера на объекте
- **_create_preview_sticker() -> void** - создает стикер, который показывает как будет выглядить стикер при его размещении на объекте
- **_update_preview() -> void** - меняет положения `preview_sticker` на объекте

### For debug

- var **sq** `Node3D` - ссылка на куб
- var **ci** `Node3D` - ссылка на сферу
- var **cy** `Node3D` - ссылкса на цилиндр
- var **figures** `Array` - хранит все вохможные объекты для размещения на сцене
- var **index_fig** `int` - количество объектов в **figures**
- const **PosObject** `Vector3` - вектор, по которому появляется объект на сцене
- func **_change_figure() -> void** - при нажатии на клавишу, меняет объекте на сцене беря их по кругу из массива **figures**

func _safe_look_at(node: Node3D, pos: Vector3, normal: Vector3) -> void:
	var up = Vector3.UP

		up = Vector3.RIGHT

func _rotate_object(axis, degrees) -> void:
	if object == null || is_rotating:
		return
	is_rotating = true

	var from = object.global_transform
	var tween = create_tween()
	tween.set_trans(Tween.TRANS_LINEAR).set_ease(Tween.EASE_OUT)

	tween.tween_method(func(progress):
		var rotated = from.rotated(axis, deg_to_rad(degrees) * progress)
		object.global_transform = rotated,
		0.0, 
		1.0, 
		ROTATE_TIME
	)

	tween.tween_callback(func():
		is_rotating = false
	)

	var ray = PhysicsRayQueryParameters3D.create(start, end)
	ray.collide_with_bodies = true
	ray.collide_with_areas = true

	var space = get_world_3d().direct_space_state
	var result = space.intersect_ray(ray)

func switch_to_scene():
	var data = read_db()
	G.quest_list = data[0]
	get_tree().change_scene_to_file(QuestScene)

func read_db():
	var data = SQLite.new()
	data.path = "res://src/DataBase/db.db"
	data.open_db()
	var result = data.query("
	SELECT 
    tasks.*,
    
    s_back.id    AS back_id,
    s_back.sticker  AS back_path,
    
    s_front.id   AS front_id,
    s_front.sticker AS front_path,
    
    s_right.id   AS right_id,
    s_right.sticker AS right_path,
    
    s_left.id    AS left_id,
    s_left.sticker  AS left_path,
    
    t.id         AS truck_id,
    t.truck       AS truck_name,
    
    th.id        AS theme_id,
    th.theme      AS theme_name
	
	FROM tasks
	LEFT JOIN stickers AS s_back  ON s_back.id  = tasks.back
	LEFT JOIN stickers AS s_front ON s_front.id = tasks.front
	LEFT JOIN stickers AS s_right ON s_right.id = tasks.right
	LEFT JOIN stickers AS s_left  ON s_left.id  = tasks.left

	LEFT JOIN trucks   AS t  ON t.id  = tasks.truck
	LEFT JOIN themes   AS th ON th.id = tasks.theme;
	")
	
	var return_list = []
	
	if result:
		for row in data.query_result:
			return_list.append(row)
	else:
		push_error("Error: query was failed " + data.error_message)

	data.close_db()
	return return_list

extends Panel

signal sticker_selected(id: String, texture_path: String)

@export var container_node: Node3D

const STICKERS_DATA = [
	{"id": "5.2", "path": "res://src/Textures/5.2.png", "label": "Class 5.2"},
	{"id": "1.4", "path": "res://src/Textures/1.4.png", "label": "UN 1.4"}
]

@onready var container_box = $ScrollContainer/HBoxContainer

func _ready() -> void:
	# Начальное состояние
	visible = false
	_populate_stickers()

func show_panel() -> void:
	if visible: return
	
	visible = true
	var screen_h = get_viewport_rect().size.y
	var target_y = screen_h - size.y - 10 # 10px отступ
	
	position.y = screen_h + 10 # Старт снизу
	
	var tween = create_tween()
	tween.set_ease(Tween.EASE_OUT).set_trans(Tween.TRANS_BACK)
	tween.tween_property(self, "position:y", target_y, 0.3)

func hide_panel() -> void:
	if not visible: return
	
	var screen_h = get_viewport_rect().size.y
	var target_y = screen_h + 10
	
	var tween = create_tween()
	tween.set_ease(Tween.EASE_IN).set_trans(Tween.TRANS_CUBIC)
	tween.tween_property(self, "position:y", target_y, 0.25)
	tween.tween_callback(func(): visible = false)

func _populate_stickers() -> void:
	for child in container_box.get_children():
		child.queue_free()
		
	for data in STICKERS_DATA:
		var btn = TextureButton.new()
		btn.custom_minimum_size = Vector2(110, 110)
		btn.ignore_texture_size = true
		btn.stretch_mode = TextureButton.STRETCH_KEEP_ASPECT_CENTERED
		
		# Загрузка текстуры (используем placeholder если нет файла)
		if ResourceLoader.exists(data.path):
			btn.texture_normal = load(data.path)
		
		btn.pressed.connect(_on_sticker_button_pressed.bind(data.id, data.path))
		
		# Добавляем подпись
		var lbl = Label.new()
		lbl.text = data.label
		lbl.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		lbl.vertical_alignment = VERTICAL_ALIGNMENT_BOTTOM
		lbl.set_anchors_preset(Control.PRESET_FULL_RECT)
		lbl.add_theme_color_override("font_color", Color.WHITE)
		lbl.add_theme_font_size_override("font_size", 14)
		btn.add_child(lbl)
		
		container_box.add_child(btn)

func _on_sticker_button_pressed(id: String, path: String) -> void:
	emit_signal("sticker_selected", id, path)
	if container_node and container_node.has_method("apply_sticker"):
		container_node.apply_sticker(path)
